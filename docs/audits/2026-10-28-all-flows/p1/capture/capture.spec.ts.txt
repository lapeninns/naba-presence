// p1 before/after capture: sign-in, auth, invites, entry, legal, shared report.
//
// Context options are tests/visual/shoot.mjs's own (viewport height 860,
// reducedMotion, hasTouch + isMobile below 768, Chromium, colorScheme for the
// theme) at the repo convention: 390 light, 390 dark, 1280 light. File names
// follow shoot.mjs too: <state>-chromium-<theme>-<width>.png.
//
// Every API the pages call is stubbed with page.route, so no form reaches a
// real auth provider, invitation or mailer. The only real writes are the
// owner's share link for F11, made through the share sheet exactly as
// tests/e2e/reports.spec.ts does, on the throwaway database. Signed-in invite
// states use the fixture owner's cookie; the session DELETE is stubbed so the
// fixture session is never destroyed.
//
//   CAPTURE_OUT=before|after [CAPTURE_MFA=1] [CAPTURE_ONLY=f05] \
//     node scripts/run-test-command.mjs e2e pnpm exec playwright test \
//     --config .omo/p1-capture/capture.config.ts
import { mkdirSync } from "node:fs"
import { resolve } from "node:path"

import {
  expect,
  test,
  type Browser,
  type BrowserContext,
  type Page,

} from "@playwright/test"

import { pageSettled } from "../../tests/e2e/helpers/settled"

const OUT = resolve(
  "docs/audits/2026-10-28-all-flows/p1/evidence",
  process.env.CAPTURE_OUT ?? "before",
  "png"
)
mkdirSync(OUT, { recursive: true })
const ONLY = process.env.CAPTURE_ONLY
const MFA = process.env.CAPTURE_MFA === "1"

const VARIANTS = [
  { width: 390, theme: "light" },
  { width: 390, theme: "dark" },
  { width: 1280, theme: "light" },
] as const

type Setup = (page: Page, context: BrowserContext) => Promise<void>

async function newContext(
  browser: Browser,
  baseURL: string,
  variant: (typeof VARIANTS)[number],
  cookie?: string
) {
  const context = await browser.newContext({
    baseURL,
    viewport: { width: variant.width, height: 860 },
    colorScheme: variant.theme,
    reducedMotion: "reduce",
    hasTouch: variant.width < 768,
    isMobile: variant.width < 768,
  })
  if (cookie) {
    const [name, value] = cookie.split("=", 2)
    await context.addCookies([{ name, value, url: baseURL }])
  }
  return context
}

async function shoot(
  browser: Browser,
  baseURL: string,
  name: string,
  setup: Setup,
  options: { cookie?: string; full?: boolean } = {}
) {
  for (const variant of VARIANTS) {
    const context = await newContext(browser, baseURL, variant, options.cookie)
    const page = await context.newPage()
    try {
      await setup(page, context)
      await page.waitForTimeout(400)
      await page.screenshot({
        path: resolve(
          OUT,
          `${name}-chromium-${variant.theme}-${variant.width}.png`
        ),
        fullPage: options.full ?? true,
      })
    } finally {
      await context.close()
    }
  }
}

const json =
  (status: number, body: unknown, headers: Record<string, string> = {}) =>
  (route: Route) =>
    route.fulfill({
      status,
      contentType: "application/json",
      headers,
      body: JSON.stringify(body),
    })
// Holds a request long enough to photograph the pending state, then lets it
// fail, so closing the context never waits on a request left open.
const holdRoute = async (route: Route) => {
  await new Promise((resolve) => setTimeout(resolve, 3000))
  await route.abort().catch(() => {})
}
const hold = () => holdRoute

const RATE_LIMITED = {
  error: "auth_rate_limited",
  message: "Too many attempts. Try again later.",
  retryable: true,
  details: { retryAfter: 840 },
}
const LIVE_INVITE = {
  organisationName: "Fixture Inns",
  email: "invited@example.test",
  role: "member",
  expiresAt: "2099-10-22T14:05:00.000Z",
}
const PASSWORD = "correct horse battery staple"

async function goto(page: Page, path: string) {
  await page.goto(path)
  await pageSettled(page)
}

async function fillSignIn(page: Page) {
  await page.getByLabel("Email address").fill("sam@example.test")
  await page.getByLabel("Password", { exact: true }).fill(PASSWORD)
}

const run = (id: string) => !ONLY || ONLY.split(",").includes(id)


test("F01 sign in", async ({ browser, baseURL }) => {
  test.skip(MFA || !run("f01"))
  const b = baseURL!
  await shoot(browser, b, "f01-01-initial", (p) => goto(p, "/sign-in"))
  await shoot(browser, b, "f01-02-next", (p) =>
    goto(p, "/sign-in?next=%2Freports")
  )
  await shoot(browser, b, "f01-03-invalid-credentials", async (p) => {
    await p.route(
      "**/api/auth/password/login",
      json(401, { error: "invalid_credentials", message: "x" })
    )
    await goto(p, "/sign-in")
    await fillSignIn(p)
    await p.getByRole("button", { name: "Sign in", exact: true }).click()
    await expect(p.getByRole("main").getByRole("alert")).toBeVisible()
  })
  await shoot(browser, b, "f01-04-retry-pending", async (p) => {
    let calls = 0
    await p.route("**/api/auth/password/login", (route) =>
      calls++ === 0
        ? json(401, { error: "invalid_credentials", message: "x" })(route)
        : holdRoute(route)
    )
    await goto(p, "/sign-in")
    await fillSignIn(p)
    await p.getByRole("button", { name: "Sign in", exact: true }).click()
    await expect(p.getByRole("main").getByRole("alert")).toBeVisible()
    await p.getByRole("button", { name: "Sign in", exact: true }).click()
    await expect(p.getByRole("button", { name: /Signing in/ })).toBeVisible()
  })
  await shoot(browser, b, "f01-05-network", async (p) => {
    await p.route("**/api/auth/password/login", (route) =>
      route.abort("internetdisconnected")
    )
    await goto(p, "/sign-in")
    await fillSignIn(p)
    await p.getByRole("button", { name: "Sign in", exact: true }).click()
    await expect(p.getByRole("main").getByRole("alert")).toBeVisible()
  })
  await shoot(browser, b, "f01-06-rate-limited", async (p) => {
    await p.route(
      "**/api/auth/password/login",
      json(429, RATE_LIMITED, { "retry-after": "840" })
    )
    await goto(p, "/sign-in")
    await fillSignIn(p)
    await p.getByRole("button", { name: "Sign in", exact: true }).click()
    await expect(p.getByRole("main").getByRole("alert")).toBeVisible()
  })
  await shoot(browser, b, "f01-07-email-link-failed", (p) =>
    goto(p, "/sign-in?status=invalid_email_link&email=error")
  )
  await shoot(browser, b, "f01-08-invite-create-mode", (p) =>
    goto(p, "/sign-in?invite=fixture-invite-token&mode=create-account")
  )

    )
    await goto(p, `/confirm-email?token_hash=${token}`)

    // capture stops the navigation after React's own submit handler has run
    // (a window listener fires after the root's). Capture-only: the page

    await p.evaluate(() =>
      window.addEventListener("submit", (event) => event.preventDefault())
    )
    await p
      .getByRole("button", { name: /^Confirm/ })
      .click({ noWaitAfter: true })
    await p.waitForTimeout(600)
  })

test("F05 two-step verification", async ({ browser, baseURL }) => {
  test.skip(!MFA || !run("f05"))
  const b = baseURL!
  const ONE = {
    factors: [{ id: "factor-1", friendlyName: "Phone" }],
    expiresAt: "2099-10-28T10:05:00.000Z",
  }
  const TWO = {
    factors: [
      { id: "factor-1", friendlyName: "Phone" },
      { id: "factor-2", friendlyName: "Tablet" },
    ],
    expiresAt: "2099-10-28T10:05:00.000Z",
  }
  const EXPIRED = {
    error: "mfa_ticket_invalid",
    message: "Your sign-in has expired. Sign in again.",
  }
  const open = async (p: Page, factors: unknown = ONE) => {
    await p.route("**/api/auth/mfa/factors", json(200, factors))
    await goto(p, "/sign-in/verify?next=%2Freports")
  }
  const verify = (p: Page) =>
    p.getByRole("button", { name: "Verify and sign in" }).click()
  await shoot(browser, b, "f05-01-ready", (p) => open(p))
  await shoot(browser, b, "f05-02-two-apps", (p) => open(p, TWO))
  await shoot(browser, b, "f05-03-wrong-code", async (p) => {
    await p.route(
      "**/api/auth/mfa/verify",
      json(400, {
        error: "mfa_code_invalid",
        message: "That code is not valid.",
        fieldErrors: { code: "That code is not valid." },
        details: { attemptsRemaining: 4 },
      })
    )
    await open(p, TWO)
    await p.keyboard.type("123456")
    await verify(p)
    await expect(p.getByText(/did not match/)).toBeVisible()
  })
  await shoot(browser, b, "f05-04-expired", async (p) => {
    await p.route("**/api/auth/mfa/verify", json(401, EXPIRED))
    await open(p)
    await p.keyboard.type("123456")
    await verify(p)
    await expect(p.getByText(/expired/).first()).toBeVisible()
  })
  await shoot(browser, b, "f05-05-load-failed", async (p) => {
    await p.route(
      "**/api/auth/mfa/factors",
      json(500, { error: "internal_error", message: "x" })
    )
    await goto(p, "/sign-in/verify")
    await expect(p.getByRole("main").getByRole("alert")).toBeVisible()
  })
  await shoot(browser, b, "f05-06-rate-limited", async (p) => {
    await p.route(
      "**/api/auth/mfa/verify",
      json(
        429,
        { ...RATE_LIMITED, details: { retryAfter: 600 } },
        {
          "retry-after": "600",
        }
      )
    )
    await open(p)
    await p.keyboard.type("123456")
    await verify(p)
    await expect(p.getByText(/Too many attempts/)).toBeVisible()
  })
  await shoot(browser, b, "f05-07-leaving", async (p) => {
    await p.route("**/api/auth/mfa/verify", (route) =>
      route.request().method() === "DELETE" ? holdRoute(route) : route.continue()
    )
    await open(p)
    await p.getByRole("button", { name: "Use a different account" }).click()
    await p.waitForTimeout(600)
  })
})

test("F08 accept invitation", async ({ browser, baseURL }) => {
  test.skip(MFA || !run("f08"))
  const b = baseURL!
  const state = await readJourneyState()
  const ownerEmail = await (async () => {
    const context = await newContext(browser, b, VARIANTS[0], state.cookie)
    const response = await context.request.get("/api/session")
    const email = (await response.json())?.session?.email as string
    await context.close()
    return email
  })()
  const INVITE = /\/api\/invitations\/[^/?]+(\?.*)?$/
  const lookup = (
    p: Page,
    handler: (route: Route) => unknown,
    post?: (route: Route) => unknown
  ) =>
    p.route(INVITE, (route) =>
      route.request().method() === "GET"
        ? handler(route)
        : post
          ? post(route)
          : route.abort()
    )
  // Never let a capture end the fixture owner's session.
  const guardSession = (p: Page) =>
    p.route("**/api/session**", (route) =>
      route.request().method() === "DELETE"
        ? holdRoute(route)
        : route.continue()
    )
  const path = "/invite/fixture-invite-token"
  await shoot(browser, b, "f08-01-loading", async (p) => {
    await lookup(p, hold())
    await p.goto(path)
    await p.waitForTimeout(800)
  })
  await shoot(browser, b, "f08-02-create", async (p) => {
    await lookup(p, json(200, LIVE_INVITE))
    await goto(p, path)
  })
  await shoot(browser, b, "f08-03-toggled-sign-in", async (p) => {
    await lookup(p, json(200, LIVE_INVITE))
    await goto(p, path)
    await p.getByRole("button", { name: "Switch to sign in" }).click()
  })
  await shoot(browser, b, "f08-04-lookup-failed", async (p) => {
    await lookup(p, json(500, { error: "internal_error", message: "x" }))
    await goto(p, path)
    await expect(p.getByRole("button", { name: "Try again" })).toBeVisible()
  })
  await shoot(browser, b, "f08-05-expired", async (p) => {
    await lookup(p, json(410, { error: "invitation_expired", message: "x" }))
    await goto(p, path)
  })
  await shoot(
    browser,
    b,
    "f08-06-used-signed-in",
    async (p) => {
      await guardSession(p)
      await lookup(
        p,
        json(409, { error: "invitation_already_used", message: "x" })
      )
      await goto(p, path)
    },
    { cookie: state.cookie }
  )
  await shoot(
    browser,
    b,
    "f08-07-match",
    async (p) => {
      await guardSession(p)
      await lookup(p, json(200, { ...LIVE_INVITE, email: ownerEmail }))
      await goto(p, path)
    },
    { cookie: state.cookie }
  )
  await shoot(
    browser,
    b,
    "f08-08-accepting",
    async (p) => {
      await guardSession(p)
      await lookup(p, json(200, { ...LIVE_INVITE, email: ownerEmail }), hold())
      await goto(p, path)
      await p.getByRole("button", { name: "Accept invitation" }).click()
      await p.waitForTimeout(400)
    },
    { cookie: state.cookie }
  )
  await shoot(
    browser,
    b,
    "f08-09-mismatch",
    async (p) => {
      await guardSession(p)
      await lookup(p, json(200, LIVE_INVITE))
      await goto(p, path)
    },
    { cookie: state.cookie }
  )
})
